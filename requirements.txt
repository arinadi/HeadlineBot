# requirements.txt — full GPU install: the CPU set plus local Whisper transcription.
# torch is not listed: Colab/Kaggle GPU images ship it, and main.py only uses it
# to check CUDA (without it the bot falls back to Gemini).
-r requirements_cpu.txt
# Upper bound: a faster-whisper major release may change transcribe()/WhisperModel.
faster-whisper<2
# av 19 removed av.open(metadata_errors=...), which faster-whisper 1.2.1 still passes; drop once faster-whisper supports av 19 (SYSTRAN/faster-whisper#1492)
av<19
# CTranslate2 4.x (faster-whisper's engine) is built for CUDA 12 and dlopens libcublas.so.12 / cuDNN 9,
# but newer torch on Colab ships CUDA 13 libs only. Bring our own CUDA 12 copies;
# headlinebot/cuda_libs.py preloads them because LD_LIBRARY_PATH can't change after start.
nvidia-cublas-cu12<13
nvidia-cudnn-cu12==9.*
